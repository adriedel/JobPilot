# Unslop-Durchgang Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Das JobPilot-Frontend nach den unslop-Regeln überarbeiten: ruhige Tokens, Navy als einziger Akzent, Iosevka Medium, Status-Verteilung und Stufen-Leiste als Highlights, ohne Funktionen zu ändern.

**Architecture:** Zuerst ersetzt eine Token-Ebene in `src/App.css` die bestehenden Werte, die Variablennamen bleiben erhalten. Danach laufen die Durchgänge über alle CSS-Dateien. Jeder Durchgang wird von einer Prüfung in `scripts/unslop-check.mjs` gemessen (erst rot, dann grün). Neue Bausteine (`StatusTrack`, umgebaute `StatCards`) werden testgetrieben mit Playwright gebaut.

**Tech Stack:** React 19, Vite 7, plain CSS mit Custom Properties, Playwright 1.65.0, Node 20+.

**Spec:** `docs/superpowers/specs/2026-11-11-unslop-design.md`

### Task 1: Prüf-Script für Slop-Merkmale

**Files:**
- Create: `JobPilot/frontend/scripts/unslop-check.mjs`
- Modify: `JobPilot/frontend/package.json` (Script `check:unslop`)

**Interfaces:**
- Produces: `npm run check:unslop` (alle Prüfungen) und `node scripts/unslop-check.mjs --only <name>`. Namen: `gradients`, `hex`, `coloredShadow`, `weight`, `uppercase`, `transitionAll`, `hoverMotion`, `spacing`, `fontSize`, `radius`. Exit-Code 1, wenn eine ausgewählte Prüfung Treffer hat. Ausgabe pro Treffer: `datei:zeile  [check]  zeileninhalt`.

- [ ] **Step 1: Script schreiben**

```js
// scripts/unslop-check.mjs
// Scannt src/**/*.css auf Merkmale aus unslop (github.com/heydamir/unslop, CC BY 4.0).
//   node scripts/unslop-check.mjs              → alle Prüfungen
//   node scripts/unslop-check.mjs --only hex   → eine Prüfung
import { readdirSync, readFileSync, statSync } from "node:fs";
import { join, relative, basename } from "node:path";
import { fileURLToPath } from "node:url";

const ROOT = fileURLToPath(new URL("../src", import.meta.url));
const TOKEN_FILE = "App.css";
const FONT_SIZES = [12, 13, 14, 16, 18, 20, 28, 36, 56];
const RADII = [0, 4, 6, 8, 12, 16, 999];

function walk(dir) {
  return readdirSync(dir).flatMap((name) => {
    const full = join(dir, name);
    return statSync(full).isDirectory() ? walk(full) : full.endsWith(".css") ? [full] : [];
  });
}

const px = (value) => {
  const m = value.match(/^(-?[\d.]+)(px|rem)$/);
  if (!m) return null;
  return m[2] === "rem" ? Number(m[1]) * 16 : Number(m[1]);
};

const checks = {
  gradients: (line) => /(linear|radial|conic)-gradient/.test(line),
  hex: (line, file) => basename(file) !== TOKEN_FILE && /#[0-9a-fA-F]{3,8}\b/.test(line),
  coloredShadow: (line) => {
    if (!/shadow\s*:/.test(line)) return false;
    return [...line.matchAll(/rgba?\((\d+),\s*(\d+),\s*(\d+)/g)].some(
      ([, r, g, b]) => !(r === g && g === b),
    );
  },
  weight: (line) => /font-weight\s*:\s*(700|800|900|bold)/.test(line),
  uppercase: (line) => /text-transform\s*:\s*uppercase/.test(line),
  transitionAll: (line) => /transition\s*:\s*all\b/.test(line),
  hoverMotion: () => false, // blockbasiert, siehe unten
  spacing: (line) => {
    const m = line.match(/^\s*(padding|margin|gap|row-gap|column-gap)(-[a-z]+)?\s*:\s*([^;]+);/);
    if (!m) return false;
    return m[3].split(/\s+/).some((v) => {
      const n = px(v);
      return n !== null && Math.abs(n) > 2 && Math.abs(n) % 4 !== 0;
    });
  },
  fontSize: (line) => {
    const m = line.match(/font-size\s*:\s*([\d.]+(px|rem))/);
    return m ? !FONT_SIZES.includes(px(m[1])) : false;
  },
  radius: (line) => {
    const m = line.match(/border-radius\s*:\s*([^;]+);/);
    if (!m) return false;
    return m[1].split(/\s+/).some((v) => {
      if (v === "50%" || v.startsWith("var(") || v === "inherit") return false;
      const n = px(v);
      return n !== null && !RADII.includes(n);
    });
  },
};

const only = process.argv.includes("--only") ? process.argv[process.argv.indexOf("--only") + 1] : null;
const active = only ? [only] : Object.keys(checks);
if (only && !checks[only]) {
  console.error(`Unbekannte Prüfung: ${only}. Verfügbar: ${Object.keys(checks).join(", ")}`);
  process.exit(2);
}

const hits = [];
for (const file of walk(ROOT)) {
  const text = readFileSync(file, "utf8");
  const lines = text.split("\n");
  lines.forEach((line, i) => {
    for (const name of active) {
      if (checks[name](line, file)) hits.push(`${relative(ROOT, file)}:${i + 1}  [${name}]  ${line.trim()}`);
    }
  });
  if (active.includes("hoverMotion")) {
    for (const block of text.matchAll(/([^{}]*:hover[^{}]*)\{([^}]*)\}/g)) {
      if (/transform\s*:(?!\s*none)/.test(block[2])) {
        const lineNo = text.slice(0, block.index).split("\n").length;
        hits.push(`${relative(ROOT, file)}:${lineNo}  [hoverMotion]  ${block[1].trim()}`);
      }
    }
  }
}

for (const hit of hits) console.log(hit);
const counts = Object.fromEntries(active.map((n) => [n, hits.filter((h) => h.includes(`[${n}]`)).length]));
console.log("\n" + Object.entries(counts).map(([n, c]) => `${c === 0 ? "✓" : "✗"} ${n}: ${c}`).join("\n"));
process.exit(hits.length ? 1 : 0);
```

```jsx
// src/components/StatusTrack/StatusTrack.jsx
import { useTranslation } from "react-i18next";
import { STATUS } from "../../utils/constants";
import "./StatusTrack.css";

const STEPS = [STATUS.IN_PLANUNG, STATUS.BEWORBEN, STATUS.STUFE_WEITER, STATUS.ANGENOMMEN];
const OUTCOMES = [STATUS.ABGELEHNT, STATUS.KEINE_ANTWORT];

/** Wie viele der 4 Stufen erreicht sind. Abgelehnt/keine Antwort: mindestens "beworben". */
export function reachedSteps(status) {
  if (OUTCOMES.includes(status)) return 2;
  const index = STEPS.indexOf(status);
  return index === -1 ? 0 : index + 1;
}

function StatusTrack({ status }) {
  const { t } = useTranslation();
  const reached = reachedSteps(status);
  const outcome = OUTCOMES.includes(status) ? status : null;

  return (
    <div
      className="status-track"
      data-status={status}
      role="img"
      aria-label={t(`status.${status}`)}
    >
      {STEPS.map((step, index) => (
        <span
          key={step}
          className="status-track-step"
          data-reached={index < reached ? "true" : "false"}
        />
      ))}
      {outcome && <span className="status-track-end" data-outcome={outcome} />}
    </div>
  );
}

export default StatusTrack;
```
